#### Remoe Latext characters: ####
find . -type f -name "*.json" -exec sed -i.bak -E '
  s/\$//g;
  s/\\+text\{([^{}]*)\}/\1/g;
  s/_\{?([0-9]+)\}?/\1/g;
  s/\\+rightarrow/->/g;
  s/\\+uparrow/gas released ↑/g;
  s/\\+downarrow/precipitate formed ↓/g;
  s/\\([A-Za-z])/\1/g
' {} \;




#### Validate Json ####
find . -type f -name "*.json" -exec powershell.exe -Command "try { Get-Content -Raw '{}' | ConvertFrom-Json > \$null; Write-Host 'OK: {}' } catch { Write-Host 'INVALID: {}'; Write-Host \$_.Exception.Message }" \;

## Delete Backup Json ##
find . -name "*.json.bak" -delete
